cmake_minimum_required(VERSION 3.16)
project(SignalFlight VERSION 1.8.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SF_SANITIZE_ADDRESS "Enable AddressSanitizer+UBSan" OFF)
option(SF_SANITIZE_THREAD "Enable ThreadSanitizer (separate build, GCC/Clang)" OFF)
option(SF_BUILD_TESTS "Build self-contained C++ tests" ON)
find_package(Threads REQUIRED)
add_library(signalflight STATIC src/protocol.cpp src/pipeline.cpp src/analysis.cpp src/recorder.cpp src/engine.cpp)
target_include_directories(signalflight PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(signalflight PUBLIC Threads::Threads)
if(WIN32)
  target_link_libraries(signalflight PUBLIC ws2_32 shell32)
endif()
if(MSVC)
  target_compile_options(signalflight PRIVATE /W4)
else()
  target_compile_options(signalflight PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(SF_SANITIZE_ADDRESS AND MSVC)
  message(FATAL_ERROR "SF_SANITIZE_ADDRESS requires GCC or Clang")
endif()
if(SF_SANITIZE_THREAD AND MSVC)
  message(FATAL_ERROR "SF_SANITIZE_THREAD requires GCC or Clang")
endif()
if(SF_SANITIZE_ADDRESS AND SF_SANITIZE_THREAD)
  message(FATAL_ERROR "AddressSanitizer and ThreadSanitizer must be separate builds")
endif()
if(SF_SANITIZE_ADDRESS)
  target_compile_options(signalflight PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(signalflight PUBLIC -fsanitize=address,undefined)
endif()
if(SF_SANITIZE_THREAD)
  target_compile_options(signalflight PUBLIC -fsanitize=thread -fno-omit-frame-pointer)
  target_link_options(signalflight PUBLIC -fsanitize=thread)
endif()
foreach(name sf_source sf_engine sf_replay sf_bench sf_crc_bench)
  add_executable(${name} apps/${name}.cpp)
  target_link_libraries(${name} PRIVATE signalflight)
  if(MSVC)
    target_compile_options(${name} PRIVATE /W4)
  else()
    target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
if(SF_BUILD_TESTS)
  enable_testing()
  foreach(test_name test_core test_hardening)
    add_executable(${test_name} tests/${test_name}.cpp)
    target_link_libraries(${test_name} PRIVATE signalflight)
    add_test(NAME ${test_name} COMMAND ${test_name})
  endforeach()
endif()
